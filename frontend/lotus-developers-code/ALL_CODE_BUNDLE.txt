# Stark Properties frontend and Django backend

The React/Vite frontend in this folder is wired to the Django application at the workspace root. Start Django on port 8000 and Vite on port 3000; `vite.config.ts` proxies API and static file requests to Django.

## API endpoints

- `GET /api/csrf/` sets Django's CSRF cookie for the browser session.
- `GET /api/properties/` returns the available property inventory as the `Property` shape used by the React UI.
- `POST /api/inquiries/` validates and stores consultation requests, then returns `{ "referenceId": "LD-...", "status": "received" }`.
- `POST /api/newsletter/` stores a validated email address.

Property records, consultation requests, and newsletter subscriptions can be managed in Django admin. Run `python manage.py migrate` from the workspace root to create the tables and sample residence concepts. Property photo uploads are saved to `MEDIA_ROOT` and returned as the API `image` URL.

The original AI Studio package omitted its photography. Blank property image URLs use the shared architectural SVG in the Django `static/images` folder. Replace contact placeholders and sample concept content with the correct Stark Properties information before publishing.

# Sample property image sources

The seeded property photos are linked from Unsplash's CDN. They are visual references for the sample residence concepts and do not depict verified Stark Properties inventory. Replace them with approved property photography in Django admin.

| Sample residence | Photo source |
| --- | --- |
| The Verdant House | [Modern villa collection](https://unsplash.com/s/photos/modern-villa) |
| The Courtyard Residence | [Modern house with large windows and sunroom](https://unsplash.com/photos/modern-house-with-large-windows-and-sunroom-g_B92WafICI) |
| The Skyline Collection | [Modern apartment building with balconies](https://unsplash.com/photos/modern-apartment-building-with-balconies-against-bright-sky-gjhCscBW1t4) |
| The Quiet Grove | [Cabin nestled in the woods](https://unsplash.com/photos/a-cabin-nestled-in-the-woods-surrounded-by-trees-9PSUpqr0Kgg) |

Each residence also has two additional Unsplash gallery photos seeded by migration `0005_property_image_gallery`. You can replace or add gallery photos from the property's **Property gallery images** inline section in Django admin. Gallery photo entries support uploaded JPG, PNG, and WebP images or an external image URL.

import { InquiryFormData } from './utils/validation';
import { Property } from './data/properties';

async function responseJson<T>(response: Response): Promise<T> {
  const contentType = response.headers.get('content-type') || '';
  const body = contentType.includes('application/json')
    ? await response.json().catch(() => ({}))
    : {};
  if (!response.ok) {
    if (response.status === 403) {
      throw new Error('Django rejected the request. Refresh the page; if you opened it from another device, make sure that address is listed in CSRF_TRUSTED_ORIGINS.');
    }
    throw new Error(body.error || 'The request could not be completed. Please try again.');
  }
  return body as T;
}

export async function loadProperties(): Promise<Property[]> {
  const response = await fetch('/api/properties/');
  return responseJson<Property[]>(response);
}

export async function prepareCsrf(): Promise<void> {
  const response = await fetch('/api/csrf/');
  await responseJson<{ detail: string }>(response);
}

function csrfToken(): string {
  const cookie = document.cookie.split('; ').find((part) => part.startsWith('csrftoken='));
  return cookie ? decodeURIComponent(cookie.split('=').slice(1).join('=')) : '';
}

export async function sendInquiry(data: InquiryFormData, propertyId?: string) {
  if (!csrfToken()) await prepareCsrf();
  const response = await fetch('/api/inquiries/', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'X-CSRFToken': csrfToken() },
    body: JSON.stringify({ ...data, propertyId }),
  });
  return responseJson<{ referenceId: string; status: string }>(response);
}

export async function subscribe(email: string) {
  if (!csrfToken()) await prepareCsrf();
  const response = await fetch('/api/newsletter/', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'X-CSRFToken': csrfToken() },
    body: JSON.stringify({ email }),
  });
  return responseJson<{ status: string }>(response);
}

import React, { useState, useEffect } from 'react';
import { CheckCircle2, Phone, Mail, MessageSquare, ShieldCheck, ArrowRight } from 'lucide-react';
import { validateInquiryForm, InquiryFormData } from '../utils/validation';
import { LOCATIONS, PROPERTY_TYPES, BUDGET_RANGES, Property } from '../data/properties';
import { sendInquiry } from '../api';
import { ContactMethod } from '../utils/validation';

interface InquirySectionProps {
  preSelectedProperty?: Property | null;
  requestedContactMethod?: ContactMethod;
}

export const InquirySection: React.FC<InquirySectionProps> = ({ preSelectedProperty, requestedContactMethod }) => {
  const [formData, setFormData] = useState<InquiryFormData>({
    fullName: '',
    email: '',
    phone: '',
    preferredLocation: 'Bengaluru',
    propertyType: 'Villa',
    budget: '₹2 Crore+',
    contactMethod: 'whatsapp',
    message: '',
    consent: false,
  });

  const [errors, setErrors] = useState<Record<string, string>>({});
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [isSubmitted, setIsSubmitted] = useState(false);
  const [referenceId, setReferenceId] = useState('');
  const [submitError, setSubmitError] = useState('');

  // When a property is selected externally (e.g. from modal), pre-fill fields
  useEffect(() => {
    if (preSelectedProperty) {
      const price = preSelectedProperty.priceNumeric;
      const budget = price === null || price > 20000000
        ? '₹2 Crore+'
        : price > 10000000
          ? '₹1–2 Crore'
          : price >= 5000000
            ? '₹50 Lakh–₹1 Crore'
            : 'Under ₹50 Lakh';
      setFormData((prev) => ({
        ...prev,
        preferredLocation: preSelectedProperty.city,
        propertyType: preSelectedProperty.propertyType,
        budget,
        message: `I would like to request a private viewing consultation for ${preSelectedProperty.name} in ${preSelectedProperty.location}.`,
      }));
    }
  }, [preSelectedProperty]);

  useEffect(() => {
    if (requestedContactMethod) {
      setFormData((prev) => ({ ...prev, contactMethod: requestedContactMethod }));
    }
  }, [requestedContactMethod]);

  const handleChange = (
    e: React.ChangeEvent<HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement>
  ) => {
    const { name, value, type } = e.target;
    const checked = (e.target as HTMLInputElement).checked;

    setFormData((prev) => ({
      ...prev,
      [name]: type === 'checkbox' ? checked : value,
    }));

    // Clear inline error on edit
    if (errors[name]) {
      setErrors((prev) => {
        const next = { ...prev };
        delete next[name];
        return next;
      });
    }
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (isSubmitting) return;

    const validation = validateInquiryForm(formData);
    if (!validation.isValid) {
      setErrors(validation.errors);
      return;
    }

    setIsSubmitting(true);
    setErrors({});
    setSubmitError('');
    try {
      const result = await sendInquiry(formData, preSelectedProperty?.id);
      setReferenceId(result.referenceId);
      setIsSubmitting(false);
      setIsSubmitted(true);
    } catch (error) {
      setSubmitError(error instanceof Error ? error.message : 'We could not send your inquiry. Please try again.');
      setIsSubmitting(false);
    }
  };

  const handleReset = () => {
    setFormData({
      fullName: '',
      email: '',
      phone: '',
      preferredLocation: 'Bengaluru',
      propertyType: 'Villa',
      budget: '₹2 Crore+',
      contactMethod: 'whatsapp',
      message: '',
      consent: false,
    });
    setIsSubmitted(false);
    setReferenceId('');
    setErrors({});
    setSubmitError('');
  };

  return (
    <section id="inquiry" className="py-24 sm:py-32 bg-[#14251D] text-[#F8F9FA] relative overflow-hidden">
      {/* Subtle organic light backdrop */}
      <div className="absolute inset-0 bg-radial from-[#315844]/20 via-transparent to-transparent pointer-events-none" />

      <div className="max-w-5xl mx-auto px-6 sm:px-8 lg:px-12 relative z-10">
        {/* Section Header */}
        <div className="text-center max-w-2xl mx-auto mb-16">
          <span className="text-xs sm:text-sm font-semibold tracking-[0.25em] uppercase text-[#C6A875] block mb-3">
            PRIVATE CONSULTATION
          </span>
          <h2 className="text-3xl sm:text-5xl font-light text-white tracking-tight leading-tight [text-wrap:balance]">
            Your next chapter starts with <span className="font-normal">a conversation.</span>
          </h2>
          <p className="mt-4 text-base sm:text-lg text-[#B8BDB7] leading-relaxed [text-wrap:balance]">
            Tell us what you are looking for. Our team will help you explore the right possibilities with clarity and architectural discretion.
          </p>
        </div>

        {/* Confirmation Screen */}
        {isSubmitted ? (
          <div className="max-w-xl mx-auto glass-obsidian rounded-2xl p-8 sm:p-10 border border-white/15 text-center shadow-2xl animate-in fade-in duration-300">
            <div className="w-16 h-16 rounded-full bg-[#315844]/30 border border-[#315844] flex items-center justify-center mx-auto mb-6 text-[#C6A875]">
              <CheckCircle2 className="w-8 h-8" />
            </div>

            <span className="text-xs font-mono font-medium text-[#C6A875] tracking-widest uppercase">
              Consultation Dossier Created
            </span>
            <h3 className="text-2xl font-light text-white mt-1 mb-3">
              Thank you, {formData.fullName}.
            </h3>

            <p className="text-sm text-[#B8BDB7] leading-relaxed mb-6">
              Your inquiry reference is <span className="font-mono text-white font-medium">{referenceId}</span>. We have recorded your preferences for a {formData.propertyType} in {formData.preferredLocation}.
            </p>

            <div className="p-4 rounded-xl bg-[#202522] border border-white/5 text-xs text-[#B8BDB7] text-left space-y-2 mb-6">
              <div className="flex justify-between">
                <span>Preferred Contact:</span>
                <span className="text-white capitalize">{formData.contactMethod}</span>
              </div>
              <div className="flex justify-between">
                <span>Contact Details:</span>
                <span className="text-white">{formData.email} · {formData.phone}</span>
              </div>
              <div className="flex justify-between">
                <span>Budget Segment:</span>
                <span className="text-white">{formData.budget}</span>
              </div>
            </div>

            <div className="p-3 rounded-lg bg-[#202522]/60 border border-white/5 text-[11px] text-[#B8BDB7]/80 italic mb-6">
              Your request has been saved. Our team can follow up using your preferred contact method.
            </div>

            <button
              onClick={handleReset}
              className="px-6 py-2.5 rounded-full text-xs uppercase tracking-wider font-semibold text-white border border-white/20 hover:border-white/50 hover:bg-white/5 transition-all cursor-pointer"
            >
              Submit Another Inquiry
            </button>
          </div>
        ) : (
          /* Lead Capture Form */
          <div className="glass-obsidian rounded-3xl p-6 sm:p-10 lg:p-12 border border-white/12 shadow-2xl backdrop-blur-xl">
            <form onSubmit={handleSubmit} noValidate className="space-y-6">
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-6">
                {/* Full Name */}
                <div className="space-y-2">
                  <label htmlFor="inquiry-fullName" className="block text-xs uppercase tracking-wider font-medium text-[#F8F9FA]">
                    Full Name <span className="text-[#A83D35]">*</span>
                  </label>
                  <input
                    id="inquiry-fullName"
                    type="text"
                    name="fullName"
                    value={formData.fullName}
                    onChange={handleChange}
                    placeholder="e.g. Devika Menon"
                    aria-required="true"
                    aria-invalid={Boolean(errors.fullName)}
                    className={`w-full bg-[#202522] border rounded-xl px-4 py-3 text-sm text-[#F8F9FA] placeholder-[#626861] focus:outline-none transition-colors ${
                      errors.fullName ? 'border-[#A83D35] ring-1 ring-[#A83D35]' : 'border-white/15 focus:border-[#C6A875]'
                    }`}
                  />
                  {errors.fullName && (
                    <p className="text-xs text-[#A83D35] mt-1">{errors.fullName}</p>
                  )}
                </div>

                {/* Email Address */}
                <div className="space-y-2">
                  <label htmlFor="inquiry-email" className="block text-xs uppercase tracking-wider font-medium text-[#F8F9FA]">
                    Email Address <span className="text-[#A83D35]">*</span>
                  </label>
                  <input
                    id="inquiry-email"
                    type="email"
                    name="email"
                    value={formData.email}
                    onChange={handleChange}
                    placeholder="devika@example.com"
                    aria-required="true"
                    aria-invalid={Boolean(errors.email)}
                    className={`w-full bg-[#202522] border rounded-xl px-4 py-3 text-sm text-[#F8F9FA] placeholder-[#626861] focus:outline-none transition-colors ${
                      errors.email ? 'border-[#A83D35] ring-1 ring-[#A83D35]' : 'border-white/15 focus:border-[#C6A875]'
                    }`}
                  />
                  {errors.email && (
                    <p className="text-xs text-[#A83D35] mt-1">{errors.email}</p>
                  )}
                </div>

                {/* Phone Number with Country Code */}
                <div className="space-y-2">
                  <label htmlFor="inquiry-phone" className="block text-xs uppercase tracking-wider font-medium text-[#F8F9FA]">
                    Phone Number <span className="text-[#A83D35]">*</span>
                  </label>
                  <input
                    id="inquiry-phone"
                    type="tel"
                    name="phone"
                    value={formData.phone}
                    onChange={handleChange}
                    placeholder="+91 98450 12345"
                    aria-required="true"
                    aria-invalid={Boolean(errors.phone)}
                    className={`w-full bg-[#202522] border rounded-xl px-4 py-3 text-sm text-[#F8F9FA] placeholder-[#626861] focus:outline-none transition-colors ${
                      errors.phone ? 'border-[#A83D35] ring-1 ring-[#A83D35]' : 'border-white/15 focus:border-[#C6A875]'
                    }`}
                  />
                  {errors.phone && (
                    <p className="text-xs text-[#A83D35] mt-1">{errors.phone}</p>
                  )}
                </div>

                {/* Preferred Location */}
                <div className="space-y-2">
                  <label htmlFor="inquiry-location" className="block text-xs uppercase tracking-wider font-medium text-[#F8F9FA]">
                    Preferred Location <span className="text-[#A83D35]">*</span>
                  </label>
                  <select
                    id="inquiry-location"
                    name="preferredLocation"
                    value={formData.preferredLocation}
                    onChange={handleChange}
                    className="w-full bg-[#202522] border border-white/15 rounded-xl px-4 py-3 text-sm text-[#F8F9FA] focus:outline-none focus:border-[#C6A875] transition-colors cursor-pointer"
                  >
                    {LOCATIONS.filter((l) => l !== 'All Locations').map((loc) => (
                      <option key={loc} value={loc} className="bg-[#202522] text-[#F8F9FA]">
                        {loc}
                      </option>
                    ))}
                  </select>
                </div>

                {/* Property Typology */}
                <div className="space-y-2">
                  <label htmlFor="inquiry-propertyType" className="block text-xs uppercase tracking-wider font-medium text-[#F8F9FA]">
                    Residence Typology <span className="text-[#A83D35]">*</span>
                  </label>
                  <select
                    id="inquiry-propertyType"
                    name="propertyType"
                    value={formData.propertyType}
                    onChange={handleChange}
                    className="w-full bg-[#202522] border border-white/15 rounded-xl px-4 py-3 text-sm text-[#F8F9FA] focus:outline-none focus:border-[#C6A875] transition-colors cursor-pointer"
                  >
                    {PROPERTY_TYPES.filter((t) => t !== 'All Residences').map((type) => (
                      <option key={type} value={type} className="bg-[#202522] text-[#F8F9FA]">
                        {type}
                      </option>
                    ))}
                  </select>
                </div>

                {/* Approximate Budget */}
                <div className="space-y-2">
                  <label htmlFor="inquiry-budget" className="block text-xs uppercase tracking-wider font-medium text-[#F8F9FA]">
                    Approximate Budget <span className="text-[#A83D35]">*</span>
                  </label>
                  <select
                    id="inquiry-budget"
                    name="budget"
                    value={formData.budget}
                    onChange={handleChange}
                    className="w-full bg-[#202522] border border-white/15 rounded-xl px-4 py-3 text-sm text-[#F8F9FA] focus:outline-none focus:border-[#C6A875] transition-colors cursor-pointer"
                  >
                    {BUDGET_RANGES.filter((b) => b !== 'Any Budget').map((range) => (
                      <option key={range} value={range} className="bg-[#202522] text-[#F8F9FA]">
                        {range}
                      </option>
                    ))}
                  </select>
                </div>
              </div>

              {/* Preferred Contact Method */}
              <div className="space-y-2 pt-2">
                <label className="block text-xs uppercase tracking-wider font-medium text-[#F8F9FA]">
                  Preferred Method of Communication
                </label>
                <div className="grid grid-cols-3 gap-3">
                  {(['whatsapp', 'email', 'phone'] as const).map((method) => {
                    const isSelected = formData.contactMethod === method;
                    return (
                      <button
                        key={method}
                        type="button"
                        onClick={() => setFormData((prev) => ({ ...prev, contactMethod: method }))}
                        className={`py-2.5 px-4 rounded-xl border text-xs uppercase tracking-wider font-medium transition-all flex items-center justify-center gap-2 cursor-pointer ${
                          isSelected
                            ? 'bg-[#315844] border-[#315844] text-white shadow-sm'
                            : 'bg-[#202522] border-white/10 text-[#B8BDB7] hover:border-white/20'
                        }`}
                      >
                        {method === 'whatsapp' && <MessageSquare className="w-3.5 h-3.5" />}

                  id="inquiry-message"

                  rows={3}

import { Property } from '../data/properties';

export interface FilterCriteria {
  location?: string;
  category?: string;
  propertyType?: string;
  budgetRange?: string;
  searchQuery?: string;
}

export function matchesBudget(numericPrice: number | null, budgetRange: string): boolean {
  if (!budgetRange || budgetRange === 'Any Budget') {
    return true;
  }

  // Price on request is treated as bespoke luxury (matching 2 Crore+ or Any)
  if (numericPrice === null) {
    return budgetRange === '₹2 Crore+';
  }

  const fiftyLakh = 5000000;
  const oneCrore = 10000000;
  const twoCrore = 20000000;

  switch (budgetRange) {
    case 'Under ₹50 Lakh':
      return numericPrice < fiftyLakh;
    case '₹50 Lakh–₹1 Crore':
      return numericPrice >= fiftyLakh && numericPrice <= oneCrore;
    case '₹1–2 Crore':
      return numericPrice > oneCrore && numericPrice <= twoCrore;
    case '₹2 Crore+':
      return numericPrice > twoCrore;
    default:
      return true;
  }
}

export function filterProperties(properties: Property[], criteria: FilterCriteria): Property[] {
  return properties.filter((property) => {
    // 1. Location filter
    if (criteria.location && criteria.location !== 'All Locations') {
      if (property.city !== criteria.location) {
        return false;
      }
    }

    // 2. Category filter
    if (criteria.category && criteria.category !== 'All Residences') {
      if (property.category !== criteria.category) {
        return false;
      }
    }

    // 3. Property Type filter
    if (criteria.propertyType && criteria.propertyType !== 'All Residences') {
      if (property.propertyType !== criteria.propertyType) {
        return false;
      }
    }

    // 4. Budget filter
    if (criteria.budgetRange && criteria.budgetRange !== 'Any Budget') {
      if (!matchesBudget(property.priceNumeric, criteria.budgetRange)) {
        return false;
      }
    }

    // 5. Search Query (free text in name, description, location)
    if (criteria.searchQuery && criteria.searchQuery.trim() !== '') {
      const q = criteria.searchQuery.toLowerCase().trim();
      const matchName = property.name.toLowerCase().includes(q);
      const matchDesc = property.description.toLowerCase().includes(q);
      const matchLoc = property.location.toLowerCase().includes(q);
      const matchCat = property.category.toLowerCase().includes(q);
      if (!matchName && !matchDesc && !matchLoc && !matchCat) {
        return false;
      }
    }

    return true;
  });
}

export type ContactMethod = 'email' | 'phone' | 'whatsapp';

export interface InquiryFormData {
  fullName: string;
  email: string;
  phone: string;
  preferredLocation: string;
  propertyType: string;
  budget: string;
  contactMethod: ContactMethod;
  message?: string;
  consent: boolean;
}

export interface ValidationResult {
  isValid: boolean;
  errors: Record<string, string>;
}

export function isValidEmail(email: string): boolean {
  if (!email || typeof email !== 'string') return false;
  // RFC 5322 compatible regex
  const emailRegex = /^[a-zA-Z0-9.!#$%&'*+/=?^_`{|}~-]+@[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?(?:\.[a-zA-Z0-9](?:[a-zA-Z0-9-]{0,61}[a-zA-Z0-9])?)+$/;
  return emailRegex.test(email.trim());
}

export function isValidPhone(phone: string): boolean {
  if (!phone || typeof phone !== 'string') return false;
  // Must have at least 7 digits, allowing +, spaces, hyphens, parentheses
  const cleaned = phone.replace(/[^0-9]/g, '');
  return cleaned.length >= 7 && cleaned.length <= 15;
}

export function validateInquiryForm(data: Partial<InquiryFormData>): ValidationResult {
  const errors: Record<string, string> = {};

  if (!data.fullName || data.fullName.trim().length < 2) {
    errors.fullName = 'Please enter your full name (at least 2 characters).';
  }

  if (!data.email || !isValidEmail(data.email)) {
    errors.email = 'Please enter a valid email address.';
  }

  if (!data.phone || !isValidPhone(data.phone)) {
    errors.phone = 'Please provide a valid phone number (minimum 7 digits).';
  }

  if (!data.preferredLocation || data.preferredLocation.trim() === '') {
    errors.preferredLocation = 'Please select your preferred location.';
  }

  if (!data.propertyType || data.propertyType.trim() === '') {
    errors.propertyType = 'Please select a residence typology.';
  }

  if (!data.budget || data.budget.trim() === '') {
    errors.budget = 'Please select an approximate budget range.';
  }

  if (!data.consent) {
    errors.consent = 'You must agree to be contacted regarding this inquiry.';
  }

  return {
    isValid: Object.keys(errors).length === 0,
    errors
  };
}

export function validateNewsletterEmail(email: string): { isValid: boolean; error?: string } {
  if (!email || email.trim() === '') {
    return { isValid: false, error: 'Email address is required.' };
  }
  if (!isValidEmail(email)) {
    return { isValid: false, error: 'Please enter a valid email address.' };
  }
  return { isValid: true };
}
